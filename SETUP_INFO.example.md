DOMAIN=              # Например: vpn.example.com (или пусто)

# Нужен если DOMAIN задан и используется TLS-сертификат.
ACME_EMAIL=          # Например: your@email.com

## ─── ЧТО УСТАНАВЛИВАТЬ ───────────────────────────────────────

# AmneziaWG VPN (wg-easy):
# Основной VPN с обфускацией против DPI. Клиент: приложение AmneziaVPN.
# Поддерживается роутерами Keenetic (протокол 1.5/2.0 — с KeeneticOS 5.1).
INSTALL_AWG=yes

# xray через 3x-ui:
# Обход DPI для отдельных приложений. Клиенты: v2rayN, Hiddify, Streisand, Nekobox.
INSTALL_XRAY=yes

## ─── XRAY ТРАНСПОРТЫ (если INSTALL_XRAY=yes) ────────────────

# VLESS + TCP + Reality (порт 443) — РЕКОМЕНДУЕТСЯ
# Лучший обход DPI. Не требует домена. Маскируется под TLS крупных сайтов.
XRAY_REALITY=yes

# VLESS + XHTTP + Reality (порт 2083) — РЕЗЕРВ
# CDN-совместимый (Cloudflare). Новый транспорт xray 25+.
XRAY_XHTTP=yes

# VLESS + WebSocket + TLS (порт 8443) — FALLBACK
# Требует домен. WS устаревает в xray (мигрируют на XHTTP) —
# включай только для старых клиентов без поддержки XHTTP.
XRAY_WS=no

## ─── ПОРТЫ СЕРВИСОВ ──────────────────────────────────────────

# WireGuard/AmneziaWG UDP порт.
# Стандартный 51820 блокируется РКН — используй нестандартный (40000–65535).
WG_PORT=49823

# Внешний HTTPS-порт для wg-easy веб-интерфейса (через nginx).
# Доступен только с доверенных IP из TRUSTED_IP_* выше.
WG_UI_PORT=51821

# Порт панели 3x-ui (если INSTALL_XRAY=yes).
# Доступен только с доверенных IP. Не используй 2053 (стандартный — сканируется).
XRAY_PANEL_PORT=56104

## ─── SSH ПОРТ ────────────────────────────────────────────────

# Рекомендуем оставить 22. Смена порта на Ubuntu 24.04/26.04 — нетривиальна
# (использует socket activation), и не даёт реальной защиты при наличии fail2ban.
SSH_PORT=22
